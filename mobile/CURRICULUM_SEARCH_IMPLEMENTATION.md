### **يحتاج تعديل يدوي صغير:**

في ملف `SubjectLessonsScreen.js`، السطر 176-184:

**استبدل:**
```javascript
const handleLessonPress = (lesson) => {
    navigation.navigate('Classroom', {

        initialMessage: `مرحباً! سنبدأ الآن درس \"${lesson.title}\". ${subject.initialMessage}`,
```

**بـ:**
```javascript
const handleLessonPress = (lesson) => {
    // بناء الرسالة الترحيبية
    let initialMessage = `مرحباً! سنبدأ الآن درس "${lesson.title}".`;
    
    // إذا كان الدرس من المنهج الدراسي ولديه محتوى محفوظ
    if (lesson.type === 'curriculum' && lesson.curriculumContent) {
        initialMessage += `\n\nهذا الدرس من ${lesson.searchStrategy?.name || 'المنهج الرسمي'}.\n\n`;
        initialMessage += `محتوى الدرس:\n${lesson.curriculumContent}`;
    } else {
        // درس مخصص أو بدون محتوى محفوظ
        initialMessage += ` ${subject.initialMessage}`;
        if (lesson.description && lesson.description !== 'درس جديد') {
            initialMessage += `\n\nوصف الدرس: ${lesson.description}`;
        }
    }
    
    navigation.navigate('Classroom', {
        mode: 'lesson',
        initialMessage: initialMessage,
        targetItem: subject.targetItem,
        subject: subject.id,
        lessonId: lesson.id,
        lessonType: lesson.type || 'custom',
        curriculumContent: lesson.curriculumContent || null
    });
};
```
